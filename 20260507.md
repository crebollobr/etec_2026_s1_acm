```
# 1. Instala a interface gráfica leve (XFCE) e o servidor RDP
sudo apt install xfce4 xfce4-goodies xrdp -y

# 2. Informa ao sistema para usar o XFCE quando alguém fizer login via RDP
echo "xfce4-session" > ~/.xsession

# 3. (Opcional, mas recomendado) Adiciona o usuário do xrdp ao grupo de certificados para evitar erros de tela preta
sudo adduser xrdp ssl-cert

# 4. Reinicia o serviço do RDP para aplicar todas as configurações
sudo systemctl restart xrdp
```
